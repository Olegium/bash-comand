Навигация по файловой системе
    
    pwd          # Показать текущую директорию
    ls           # Список файлов и директорий
    ls -la       # Подробный список (включая скрытые файлы)
    cd /path     # Перейти в директорию
    cd ..        # Перейти на уровень выше
    cd ~         # Перейти в домашнюю директорию
    cd -         # Вернуться в предыдущую директорию

Работа с файлами и директориями

    mkdir dir    # Создать директорию
    mkdir -p path/to/dir  # Создать директорию с родительскими
    touch file   # Создать пустой файл или обновить timestamp
    cp src dest  # Копировать файл/директорию
    cp -r src dest  # Копировать рекурсивно
    mv src dest  # Переместить/переименовать
    rm file      # Удалить файл
    rm -r dir    # Удалить директорию рекурсивно
    rm -rf dir   # Принудительное удаление (осторожно!)

Просмотр содержимого файлов

    cat file     # Вывести содержимое файла
    less file    # Постраничный просмотр
    more file    # Постраничный просмотр (простой)
    head file    # Первые 10 строк
    head -n 20 file  # Первые 20 строк
    tail file    # Последние 10 строк
    tail -f file # Мониторинг файла в реальном времени

Поиск

    find /path -name "pattern"   # Найти файлы по имени
    find /path -type f           # Найти только файлы
    find /path -type d           # Найти только директории
    grep "pattern" file          # Поиск текста в файле
    grep -r "pattern" dir        # Рекурсивный поиск
    grep -i "pattern" file       # Регистронезависимый поиск
    locate filename              # Быстрый поиск по базе данных
    which command                # Найти путь к команде
    whereis command              # Найти бинарник, исходники, man

Права доступа

    chmod 755 file     # Изменить права (числовой формат)
    chmod +x file      # Сделать исполняемым
    chown user:group file  # Изменить владельца
    ls -la             # Посмотреть права

Информация о системе

    uname -a           # Информация о ядре
    hostname           # Имя хоста
    whoami             # Текущий пользователь
    id                 # ID пользователя и группы
    df -h              # Использование диска
    du -sh dir         # Размер директории
    free -h            # Использование памяти
    top                # Процессы в реальном времени
    ps aux             # Все процессы
    uptime             # Время работы системы

Сеть

    ip addr            # IP адреса (современный аналог ifconfig)
    ifconfig           # Сетевые интерфейсы (устаревший)
    ping host          # Проверка доступности
    netstat -tulnp     # Открытые порты и процессы
    ss -tulnp          # Альтернатива netstat
    curl URL           # HTTP запросы
    wget URL           # Скачать файл
    ssh user@host      # SSH подключение
    scp file user@host:/path  # Копирование по SSH

Управление процессами

    ps aux             # Все процессы
    ps aux | grep name # Найти процесс
    kill PID           # Завершить процесс
    kill -9 PID        # Принудительно завершить
    killall name       # Завершить все процессы с именем
    bg                 # Отправить в фон
    fg                 # Вернуть из фона
    jobs               # Список фоновых задач
    nohup command &    # Запустить в фоне, не завися от терминала

Архивация

    tar -czf archive.tar.gz dir   # Создать архив
    tar -xzf archive.tar.gz       # Распаковать архив
    zip archive.zip files         # Создать ZIP
    unzip archive.zip             # Распаковать ZIP

Текст и строки

    echo "text"      # Вывести текст
    echo "text" > file   # Записать в файл (перезаписать)
    echo "text" >> file  # Добавить в файл
    wc -l file       # Количество строк
    sort file        # Сортировка
    uniq file        # Убрать дубликаты
    cut -d: -f1 file # Вырезать поля
    sed 's/old/new/g' file  # Замена текста
    awk '{print $1}' file   # Обработка текста

История и помощь

    history          # История команд
    !number          # Выполнить команду по номеру
    !!               # Повторить последнюю команду
    man command      # Руководство по команде
    command --help   # Краткая помощь

Переменные и окружение

    export VAR=value     # Экспортировать переменную
    echo $VAR            # Вывести значение переменной
    env                  # Все переменные окружения
    source ~/.bashrc     # Перезагрузить конфиг

Конвейеры и перенаправление
    
    command1 | command2      # Передать вывод одной команды другой
    command > file           # Перенаправить вывод в файл
    command >> file          # Добавить вывод в файл
    command < file           # Взять ввод из файла
    command 2> error.log     # Перенаправить ошибки
    command &> all.log       # Перенаправить всё
